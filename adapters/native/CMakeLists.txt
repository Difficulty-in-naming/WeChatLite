cmake_minimum_required(VERSION 3.24)
project(WeChatActions LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WeChatActions requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_library(actions_minhook STATIC
    third_party/minhook/src/buffer.c
    third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c
    third_party/minhook/src/hde/hde64.c)
target_include_directories(actions_minhook PUBLIC third_party/minhook/include
    PRIVATE third_party/minhook/src)
foreach(target WeChatActions ActionsSelftest WeChatQuote QuoteSelftest)
    if(target STREQUAL "WeChatActions" OR target STREQUAL "WeChatQuote")
        add_library(${target} SHARED src/actions.cpp)
    else()
        add_executable(${target} src/actions.cpp)
        target_compile_definitions(${target} PRIVATE ACTION_SELFTEST)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    endif()
    if(target STREQUAL "WeChatQuote" OR target STREQUAL "QuoteSelftest")
        target_compile_definitions(${target} PRIVATE WECHAT_QUOTE_ONLY)
    endif()
    target_include_directories(${target} PRIVATE include)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc /W4)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${target} PRIVATE actions_minhook ws2_32)
    target_link_options(${target} PRIVATE /Brepro)
endforeach()
enable_testing()
add_test(NAME actions-offline COMMAND ActionsSelftest)
add_test(NAME quote-offline COMMAND QuoteSelftest)

foreach(target WeChatMedia MediaSelftest)
    if(target STREQUAL "WeChatMedia")
        add_library(${target} SHARED src/downloads.cpp)
    else()
        add_executable(${target} src/downloads.cpp)
        target_compile_definitions(${target} PRIVATE DOWNLOAD_SELFTEST)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    endif()
    target_include_directories(${target} PRIVATE include)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc /W4)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(${target} PRIVATE actions_minhook ws2_32)
    target_link_options(${target} PRIVATE /Brepro)
endforeach()
add_test(NAME media-download-offline COMMAND MediaSelftest)

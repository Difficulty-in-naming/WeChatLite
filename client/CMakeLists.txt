    main.cpp window.cpp window.h composer.cpp composer.h
    transport.cpp transport.h messages.cpp messages.h
    attachment.cpp attachment.h voice_player.cpp voice_player.h
    connection_info.cpp connection_info.h tunnel_session.cpp tunnel_session.h
    history_browser.cpp history_browser.h announcement_dialog.cpp announcement_dialog.h
    ui_icons.qrc
    chat_preferences.cpp chat_preferences.h system_notifier.cpp system_notifier.h)

set_target_properties(wechat-lite PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "微信轻聊" MACOSX_BUNDLE_GUI_IDENTIFIER "local.wechatlite.client"
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(WIN32)
    target_sources(wechat-lite PRIVATE system_notifier.win.cpp)
    target_link_libraries(wechat-lite PRIVATE windowsapp runtimeobject ole32 shell32 propsys advapi32 uuid)
elseif(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    target_sources(wechat-lite PRIVATE system_notifier.mm)
    set_source_files_properties(system_notifier.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
    find_library(USER_NOTIFICATIONS_FRAMEWORK UserNotifications REQUIRED)
    find_library(APP_KIT_FRAMEWORK AppKit REQUIRED)
    target_link_libraries(wechat-lite PRIVATE ${FOUNDATION_FRAMEWORK} ${USER_NOTIFICATIONS_FRAMEWORK} ${APP_KIT_FRAMEWORK})
endif()
if(MSVC)
    target_compile_options(wechat-lite PRIVATE /utf-8 /W4)
endif()
